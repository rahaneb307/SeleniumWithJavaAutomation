1) Aliase an inner left Join

select c.first_name as FirstName ,o.item as Item
from  Customers c
inner join
Orders o
on c.customer_id=o.customer_id
order by 
FirstName asc


2) Full join 

select c.first_name as FirstName ,o.item as Item
from  Customers c
full join
Orders o
on c.customer_id=o.customer_id
order by 
FirstName asc

2) Right Join

select c.first_name as FirstName ,o.item as Item
from  Customers c
right join
Orders o
on c.customer_id=o.customer_id
order by 
FirstName asc
