Switching frame 
-------------------------------------------

1)driver.switchTo().frame(frame id);
2)driver.switchTo().frame(iframe name or id);
3)driver.switchTo().frame(WebElement of frame);

Example

import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;

import java.util.List;

public class FrameExample {

    public static void main(String[] args) {

        WebDriver driver = new ChromeDriver();
        driver.get("https://example.com");

        // Get all iframe elements
        List<WebElement> frames = driver.findElements(By.tagName("iframe"));

        System.out.println("Total Frames: " + frames.size());

        for (int i = 0; i < frames.size(); i++) {

            // Switch to frame
            driver.switchTo().frame(i);

            try {
                WebElement element = driver.findElement(By.id("username"));
                element.sendKeys("Balasaheb");

                System.out.println("Element found in Frame Index: " + i);

                // Switch back to parent frame
                driver.switchTo().parentFrame();
                break;

            } catch (Exception e) {
                // Element not found in this frame
                driver.switchTo().defaultContent();
            }
        }

        driver.quit();
    }
}
